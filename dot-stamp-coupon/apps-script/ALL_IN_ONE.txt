/**
 * 도트 운동 도장판 + 쿠폰 발급 웹앱 (Google Apps Script)
 *

 *  - Config  : 통과 기준 수치, 쿠폰 발급 기준 등 설정값
 */

function userLogin(name) {
  name = cleanName_(name);
  withLock_(function () {
    const sh = sheet_('USERS');
    const data = rows_('USERS');
    const idx = data.findIndex(function (r) { return String(r[0]).trim() === name; });
    const now = nowStr_();
    if (idx >= 0) sh.getRange(idx + 2, 3).setValue(now);
    else sh.appendRow([name, now, now]);
  });
  return getUserMonth(name, today_().slice(0, 7));
}

function getUserMonth(name, ym) {
  name = cleanName_(name);
  if (!/^\d{4}-\d{2}$/.test(String(ym))) ym = today_().slice(0, 7);
  const cfg = config_();
  const records = {};
  let totalApproved = 0;
  let monthApproved = 0;
  rows_('RECORDS').forEach(function (r) {
    if (String(r[1]).trim() !== name) return;
    const d = date_(r[0]);
    const st = String(r[4] || STATUS.PENDING).trim();
    if (st === STATUS.APPROVED) totalApproved++;
    if (d.slice(0, 7) === ym) {
      const rv = recValue_(r, cfg);
      records[d] = { value: rv.value, pass: rv.pass, status: st, granted: rv.granted, note: rv.note };
      if (st === STATUS.APPROVED) monthApproved++;
    }
  });
  return {
    name: name, ym: ym, today: today_(),
    threshold: cfg.PASS_THRESHOLD,
    totalApproved: totalApproved, monthApproved: monthApproved,
    records: records,
    rewards: rewards_(),
    wallets: walletMonths_().map(function (m) { return wallet_(name, m); }),
    coupons: userCoupons_(name),
  };
}

/** 도장을 소모해서 보상 쿠폰으로 교환 */
function exchangeReward(name, rewardId, ym) {
  name = cleanName_(name);
  ym = String(ym || '');
  if (walletMonths_().indexOf(ym) < 0) throw new Error('지금은 ' + ym + ' 도장을 교환할 수 없어요.');
  const reward = rewards_().filter(function (r) { return r.id === String(rewardId); })[0];
  if (!reward) throw new Error('보상을 찾을 수 없어요.');
  let coupon;
  withLock_(function () {
    const w = wallet_(name, ym);
    let cost = 0;
    if (reward.perfect) {
      if (!w.perfect) throw new Error('만근(그달 모든 날 승인)을 달성해야 받을 수 있어요.');
      if (w.claimed.indexOf(reward.id) >= 0) throw new Error('이번 달 만근 보상은 이미 받았어요.');
    } else {
      cost = reward.cost;
      if (w.balance < cost) throw new Error('도장이 부족해요. (보유 ' + w.balance + ' / 필요 ' + cost + ')');
    }
    const code = makeCouponCode_(config_().COUPON_PREFIX);
    sheet_('COUPONS').appendRow([code, name, nowStr_(), reward.id, reward.name, ym, cost, '미사용', '']);
    coupon = { code: code, rewardName: reward.name, desc: reward.desc, icon: reward.icon, ym: ym, cost: cost, balanceAfter: w.balance - cost };
  });
  const data = getUserMonth(name, ym);
  data.newCoupon = coupon;
  return data;
}

function saveRecord(name, date, value) {
  name = cleanName_(name);
  date = String(date);
  if (!/^\d{4}-\d{2}-\d{2}$/.test(date)) throw new Error('날짜 형식이 올바르지 않아요.');
  if (date > today_()) throw new Error('미래 날짜는 기록할 수 없어요.');
  const v = Number(value);
  if (String(value).trim() === '' || !isFinite(v) || v < 0 || v > 100000) throw new Error('0 ~ 100000 사이의 숫자를 입력하세요.');
  const cfg = config_();
  const auto = v >= cfg.PASS_THRESHOLD ? '통과' : '미통과';

  withLock_(function () {
    const sh = sheet_('RECORDS');
    const data = rows_('RECORDS');
    const idx = data.findIndex(function (r) { return date_(r[0]) === date && String(r[1]).trim() === name; });
    const now = nowStr_();
    if (idx >= 0) {
      const st = String(data[idx][4]).trim();
      if (st === STATUS.APPROVED || st === STATUS.REJECTED) throw new Error('관리자가 이미 처리한 기록은 수정할 수 없어요.');
      sh.getRange(idx + 2, 3, 1, 3).setValues([[v, auto, STATUS.PENDING]]);
      sh.getRange(idx + 2, 8).setValue(now);
    } else {
      sh.appendRow([date, name, v, auto, STATUS.PENDING, '', '', now]);
    }
  });
  return getUserMonth(name, date.slice(0, 7));
}

/* ───────────────────────── 미니게임 ───────────────────────── */

function saveGameScore(name, score, maxLevel) {
  name = cleanName_(name);
  score = Math.max(0, Math.min(1000000, Math.floor(Number(score) || 0)));
  maxLevel = Math.max(0, Math.min(20, Math.floor(Number(maxLevel) || 0)));
  withLock_(function () { sheet_('GAMES').appendRow([nowStr_(), name, score, maxLevel]); });
  return getGameRanking(name);
}

function getGameRanking(name) {
  const best = {};
  let plays = 0;
  rows_('GAMES').forEach(function (r) {
    const n = String(r[1]).trim();
    const s = Number(r[2]) || 0;
    if (name && n === name) plays++;
    if (!(n in best) || s > best[n]) best[n] = s;
  });
  const list = Object.keys(best).map(function (n) { return { name: n, score: best[n] }; })
    .sort(function (a, b) { return b.score - a.score; });
  return { top: list.slice(0, 10), myBest: name && best[name] != null ? best[name] : 0, myPlays: plays };
}

/** opts: { status: '대기'|'승인'|'비승인'|'전체', date: 'yyyy-MM-dd' | '', name: '' } */
function adminListRecords(auth, opts) {
  verifyAdmin_(auth);
  opts = opts || {};
  const cfg = config_();
  const status = opts.status || STATUS.PENDING;
  const date = String(opts.date || '');
  const q = String(opts.name || '').trim();
  const list = [];
  rows_('RECORDS').forEach(function (r) {
    const d = date_(r[0]);
    const n = String(r[1]).trim();
    const st = String(r[4] || STATUS.PENDING).trim();
    if (!d || !n) return;
    if (status !== '전체' && st !== status) return;
    if (date && d !== date) return;
    if (q && n.indexOf(q) < 0) return;
    const rv = recValue_(r, cfg);
    list.push({ date: d, name: n, value: rv.value, pass: rv.pass, granted: rv.granted, note: rv.note, status: st, admin: String(r[5] || ''), reviewedAt: dateTime_(r[6]) });
  });
  list.sort(function (a, b) { return a.date < b.date ? 1 : a.date > b.date ? -1 : a.name.localeCompare(b.name); });
  return { list: list.slice(0, 300), total: list.length, threshold: cfg.PASS_THRESHOLD };
}

/** decision: '승인' | '비승인' */
function adminReview(auth, date, name, decision) {
  const admin = verifyAdmin_(auth);
  if (decision !== STATUS.APPROVED && decision !== STATUS.REJECTED) throw new Error('잘못된 처리값입니다.');
  name = cleanName_(name);
  withLock_(function () {
    const sh = sheet_('RECORDS');
    const data = rows_('RECORDS');
    const idx = data.findIndex(function (r) { return date_(r[0]) === date && String(r[1]).trim() === name; });
    if (idx < 0) throw new Error('기록을 찾을 수 없어요.');
    sh.getRange(idx + 2, 5, 1, 3).setValues([[decision, admin, nowStr_()]]);
  });
  return { ok: true };
}

/** 관리자가 원하는 날짜에 합격(승인) 도장을 직접 찍어줌. 기록이 없어도 가능, 수치는 선택 */
function adminGrantStamp(auth, name, date, value, memo) {
  const admin = verifyAdmin_(auth);
  name = cleanName_(name);
  date = String(date || '');
  if (!/^\d{4}-\d{2}-\d{2}$/.test(date)) throw new Error('날짜를 선택하세요.');
  if (date > today_()) throw new Error('미래 날짜에는 도장을 줄 수 없어요.');
  const known = rows_('USERS').some(function (r) { return String(r[0]).trim() === name; });
  if (!known) throw new Error('등록되지 않은 사용자예요. (한 번 이상 로그인한 사용자만 가능)');
  const hasValue = value !== null && value !== undefined && String(value).trim() !== '';
  const v = hasValue ? Number(value) : '';
  if (hasValue && (!isFinite(v) || v < 0 || v > 100000)) throw new Error('수치는 0 ~ 100000 사이로 입력하세요.');
  const cfg = config_();
  const note = '관리자부여' + (String(memo || '').trim() ? ': ' + String(memo).trim().slice(0, 100) : '');
  let existed = false;
  withLock_(function () {
    const sh = sheet_('RECORDS');
    const data = rows_('RECORDS');
    const idx = data.findIndex(function (r) { return date_(r[0]) === date && String(r[1]).trim() === name; });
    const now = nowStr_();
    if (idx >= 0) {
      existed = true;
      const row = data[idx];
      const nv = hasValue ? v : row[2];
      const auto = nv === '' ? '관리자부여' : (Number(nv) >= cfg.PASS_THRESHOLD ? '통과' : '미통과');
      sh.getRange(idx + 2, 3, 1, 7).setValues([[nv, auto, STATUS.APPROVED, admin, now, row[7], note]]);
    } else {
      const auto = v === '' ? '관리자부여' : (v >= cfg.PASS_THRESHOLD ? '통과' : '미통과');
      sh.appendRow([date, name, v, auto, STATUS.APPROVED, admin, now, now, note]);
    }
  });
  return { ok: true, existed: existed, wallet: wallet_(name, date.slice(0, 7)) };
}

/* ───────────────────────── 내부 함수 ───────────────────────── */

let rewardsCache_ = null;
/** 기록 한 줄의 수치/통과 여부. 수치 없이 관리자가 부여한 도장은 통과로 표시 */
function recValue_(r, cfg) {
  const note = String(r[8] || '');
  const granted = note.indexOf('관리자부여') === 0;
  const empty = r[2] === '' || r[2] === null;
  const v = empty ? null : Number(r[2]);
  return { value: v, pass: empty ? granted : v >= cfg.PASS_THRESHOLD, granted: granted, note: note };
}

function rewards_() {
  if (rewardsCache_) return rewardsCache_;
  let rows = rows_('REWARDS').filter(function (r) { return String(r[0]).trim() !== ''; });
  if (!rows.length) rows = DEFAULT_REWARDS;
  rewardsCache_ = rows.map(function (r) {
    const perfect = String(r[3]).trim() === '만근';
    return { id: String(r[0]).trim(), name: String(r[1]), desc: String(r[2]), perfect: perfect, cost: perfect ? 0 : Math.max(1, Number(r[3]) || 0), icon: String(r[4] || 'ticket') };
  }).sort(function (a, b) { return (a.perfect ? 1e9 : a.cost) - (b.perfect ? 1e9 : b.cost); });
  return rewardsCache_;
}

/** 교환 가능한 달: 이번 달 + (이번 달 초 유예기간 동안) 지난달 */
function walletMonths_() {
  const t = today_();
  const months = [t.slice(0, 7)];
  if (Number(t.slice(8)) <= config_().EXCHANGE_GRACE_DAYS) months.push(prevMonth_(t.slice(0, 7)));
  return months;
}

/** 한 달 도장 지갑: 승인 도장 수 - 교환에 소모한 도장 수 */
function wallet_(name, ym) {
  const approvedDays = {};
  rows_('RECORDS').forEach(function (r) {
    if (String(r[1]).trim() !== name || String(r[4]).trim() !== STATUS.APPROVED) return;
    const d = date_(r[0]);
    if (d.slice(0, 7) === ym) approvedDays[d] = true;
  });
  const approved = Object.keys(approvedDays).length;
  let used = 0;
  const claimed = [];
  rows_('COUPONS').forEach(function (r) {
    if (String(r[1]).trim() !== name || month_(r[5]) !== ym) return;
    used += Number(r[6]) || 0;
    claimed.push(String(r[3]));
  });
  const p = ym.split('-');
  const days = new Date(Number(p[0]), Number(p[1]), 0).getDate();
  return { ym: ym, approved: approved, used: used, balance: Math.max(0, approved - used), days: days, perfect: approved >= days, claimed: claimed };
}

function userCoupons_(name) {
  return rows_('COUPONS').filter(function (r) { return String(r[1]).trim() === name; }).map(couponObj_).reverse();
}

function couponObj_(r) {
  const rw = rewards_().filter(function (x) { return x.id === String(r[3]); })[0] || {};
  return {
    code: String(r[0]), name: String(r[1]), issuedAt: dateTime_(r[2]), rewardId: String(r[3]), rewardName: String(r[4]),
    desc: rw.desc || '', icon: rw.icon || 'ticket', ym: month_(r[5]), cost: Number(r[6]) || 0,
    used: String(r[7]).trim() === '사용', usedAt: dateTime_(r[8]),
  };
}

function prevMonth_(ym) {
  const p = ym.split('-').map(Number);
  const d = new Date(p[0], p[1] - 2, 1);
  return d.getFullYear() + '-' + ('0' + (d.getMonth() + 1)).slice(-2);
}

function month_(v) {
  if (v instanceof Date) return Utilities.formatDate(v, tz_(), 'yyyy-MM');
  return String(v || '').trim().slice(0, 7);
}

function makeCouponCode_(prefix) {
  const chars = 'ABCDEFGHJKLMNPQRSTUVWXYZ23456789';
  let s = '';
  for (let i = 0; i < 8; i++) s += chars.charAt(Math.floor(Math.random() * chars.length));
  return String(prefix || 'FIT') + '-' + s.slice(0, 4) + '-' + s.slice(4);
}

function verifyAdmin_(auth) {
  const name = String((auth && auth.name) || '').trim();
  const pw = String((auth && auth.password) || '');
  if (!name || !pw) throw new Error('관리자 이름과 비밀번호를 입력하세요.');
  const ok = rows_('ADMINS').some(function (r) {
    return String(r[0]).trim() === name && String(r[1]).trim() === pw.trim() && String(r[1]).trim() !== '';
  });
  if (!ok) throw new Error('관리자 정보가 올바르지 않아요.');
  return name;
}

function cleanName_(name) {
  name = String(name || '').trim().replace(/\s+/g, ' ');
  if (!name) throw new Error('이름을 입력하세요.');
  if (name.length > 20) throw new Error('이름은 20자 이하로 입력하세요.');
  if (/^[=+\-@]/.test(name)) throw new Error('이름은 =, +, -, @ 로 시작할 수 없어요.');
  return name;
}

function ss_() {
  return SPREADSHEET_ID ? SpreadsheetApp.openById(SPREADSHEET_ID) : SpreadsheetApp.getActiveSpreadsheet();
}

function ensureSheets_() {
  Object.keys(SHEETS).forEach(function (k) { sheet_(k); });
}

const sheetCache_ = {};
function sheet_(key) {
  if (sheetCache_[key]) return sheetCache_[key];
  const def = SHEETS[key];
  const ss = ss_();
  let sh = ss.getSheetByName(def.name);
  if (!sh) {
    sh = ss.insertSheet(def.name);
    sh.getRange(1, 1, 1, def.headers.length).setValues([def.headers])
      .setFontWeight('bold').setBackground('#222034').setFontColor('#ffffff');
    sh.setFrozenRows(1);
    if (key === 'RECORDS') sh.getRange('A:A').setNumberFormat('@');
    if (key === 'ADMINS') {
      sh.getRange('B:B').setNumberFormat('@');
      sh.appendRow(['admin', '1234', '기본 관리자 - 비밀번호를 꼭 바꾸세요. 행을 추가하면 관리자도 늘어납니다.']);
    }
    if (key === 'CONFIG') sh.getRange(2, 1, DEFAULT_CONFIG.length, 3).setValues(DEFAULT_CONFIG);
    if (key === 'COUPONS') sh.getRange('F:F').setNumberFormat('@');
    if (key === 'REWARDS') {
      sh.getRange('D:D').setNumberFormat('@');
      sh.getRange(2, 1, DEFAULT_REWARDS.length, 5).setValues(DEFAULT_REWARDS.map(function (r) { return r.map(String); }));
    }
  }
  // 이전 버전 시트에 새로 추가된 열 제목 채우기
  const lastHeader = def.headers[def.headers.length - 1];
  if (String(sh.getRange(1, def.headers.length).getValue()) !== lastHeader) sh.getRange(1, 1, 1, def.headers.length).setValues([def.headers]);
  sheetCache_[key] = sh;
  return sh;
}

function rows_(key) {
  const sh = sheet_(key);
  const n = sh.getLastRow() - 1;
  if (n <= 0) return [];
  return sh.getRange(2, 1, n, SHEETS[key].headers.length).getValues();
}

function config_() {
  const o = {};
  DEFAULT_CONFIG.forEach(function (r) { o[r[0]] = r[1]; });
  rows_('CONFIG').forEach(function (r) {
    const k = String(r[0]).trim();
    if (k && r[1] !== '') o[k] = r[1];
  });
  o.PASS_THRESHOLD = Number(o.PASS_THRESHOLD) || 300;
  o.EXCHANGE_GRACE_DAYS = Math.max(0, Math.floor(Number(o.EXCHANGE_GRACE_DAYS) || 0));
  o.APP_TITLE = String(o.APP_TITLE || '도트 운동 도장판');
  return o;
}

function withLock_(fn) {
  const lock = LockService.getScriptLock();
  lock.waitLock(15000);
  try { return fn(); } finally { lock.releaseLock(); }
}

function tz_() { return Session.getScriptTimeZone() || 'Asia/Seoul'; }
function today_() { return Utilities.formatDate(new Date(), tz_(), 'yyyy-MM-dd'); }
function nowStr_() { return Utilities.formatDate(new Date(), tz_(), 'yyyy-MM-dd HH:mm:ss'); }
function date_(v) {
  if (v instanceof Date) return Utilities.formatDate(v, tz_(), 'yyyy-MM-dd');
  return String(v || '').trim();
}
function dateTime_(v) {
  if (v instanceof Date) return Utilities.formatDate(v, tz_(), 'yyyy-MM-dd HH:mm');
  return String(v || '').trim();
}

var Sprites = (function () {
  function svgFromGrid(grid, w, h, cls) {
    var r = '';
    for (var y = 0; y < h; y++) for (var x = 0; x < w; x++) {
      var c = grid[y][x];
      if (c) r += '<rect x="' + x + '" y="' + y + '" width="1.02" height="1.02" fill="' + c + '"/>';
    }
    return '<svg class="' + (cls || '') + '" viewBox="0 0 ' + w + ' ' + h + '" shape-rendering="crispEdges" xmlns="http://www.w3.org/2000/svg">' + r + '</svg>';
  }

  function fromStrings(rows, pal, cls) {
    var h = rows.length, w = rows[0].length, g = [];
    for (var y = 0; y < h; y++) { g.push([]); for (var x = 0; x < w; x++) g[y].push(pal[rows[y][x]] || null); }
    return svgFromGrid(g, w, h, cls);
  }

  function line(x0, y0, x1, y1, cb) {
    var dx = Math.abs(x1 - x0), sx = x0 < x1 ? 1 : -1, dy = -Math.abs(y1 - y0), sy = y0 < y1 ? 1 : -1, err = dx + dy;
    for (;;) {
      cb(x0, y0);
      if (x0 === x1 && y0 === y1) break;
      var e2 = 2 * err;
      if (e2 >= dy) { err += dy; x0 += sx; }
      if (e2 <= dx) { err += dx; y0 += sy; }
    }
  }

  /* 승인(PASS) / 비승인(FAIL) 도장: 20x20 도트 */
  function stamp(type) {
    var N = 20, c = (N - 1) / 2, g = [];
    var pass = type === 'pass';
    var ring = pass ? '#1f7a3a' : '#a8182a';

      for (var x = 0; x < N; x++) {
        var d = Math.sqrt((x - c) * (x - c) + (y - c) * (y - c));
        var col = null;
        if (d <= 9.7) col = d > 8.4 ? ring : (d > 7.4 ? ring2 : fill);
        // 도장 잉크가 덜 묻은 느낌의 구멍

        g[y].push(col);

      line(13, 6, 6, 13, function (x, y) { put(x, y); put(x + 1, y); });
    }

      else if (lv === 8 && (x + y) % 5 === 0 && (x - y + 50) % 5 === 0) col = dark; // 파인애플
      var hx = dx + r * 0.42, hy = dy + r * 0.42;
      if (d <= r - 1.1 && hx * hx + hy * hy < (r * 0.22) * (r * 0.22)) col = [255, 255, 255]; // 하이라이트
      var i = (y * D + x) * 4;
      img.data[i] = col[0]; img.data[i + 1] = col[1]; img.data[i + 2] = col[2]; img.data[i + 3] = 255;
    }
    x2.putImageData(img, 0, 0);
    // 귀여운 얼굴
    if (r >= 6) {
      x2.fillStyle = '#1a1c2c';
      var e = Math.max(1, Math.round(r / 8)), ex = Math.round(r * 0.38), ey = Math.round(r * 0.95);
      x2.fillRect(r - ex - e, ey, e, e + (r > 14 ? 1 : 0));
      x2.fillRect(r + ex, ey, e, e + (r > 14 ? 1 : 0));
      x2.fillRect(r - e, ey + e + 1, e * 2, e);
    }
    // 꼭지
    x2.fillStyle = '#257179';
    var s = Math.max(1, Math.round(r / 6));
    x2.fillRect(r - Math.ceil(s / 2), 0, s, s + 1);

  }

  function init(opts) {

  function rnd() { return Math.floor(Math.random() * 5); }
  function reset() {
    fruits = []; score = 0; maxLv = 0; dangerT = 0; over = false; cooldown = 0; idSeq = 0;
    aimX = W / 2; cur = rnd(); next = rnd();
    onScore && onScore(score);
    drawNext();
    document.getElementById('g-over').classList.add('hidden');

  function drawNext() {

  }

  function drop() {
    if (cooldown > 0 || over) return;
    var r = FRUITS[cur].r, x = clampX(aimX, r);
    fruits.push({ id: idSeq++, x: x, y: HOLD_Y, px: x, py: HOLD_Y - 0.3, r: r, lv: cur, born: performance.now() });
    cur = next; next = rnd(); drawNext();
    cooldown = 28;
  }
  function clampX(x, r) { return Math.max(r + 1, Math.min(W - r - 1, x)); }

  function step() {
    var i, j, a, b;
    for (var s = 0; s < SUB; s++) {
      for (i = 0; i < fruits.length; i++) {
        a = fruits[i];
        var vx = (a.x - a.px) * DAMP, vy = (a.y - a.py) * DAMP;
        vx = Math.max(-MAXV, Math.min(MAXV, vx)); vy = Math.max(-MAXV, Math.min(MAXV, vy));
        a.px = a.x; a.py = a.y;
        a.x += vx; a.y += vy + G;
      }
      var merges = [];
      for (var it = 0; it < 3; it++) {
        for (i = 0; i < fruits.length; i++) {
          a = fruits[i];
          for (j = i + 1; j < fruits.length; j++) {
            b = fruits[j];
            var dx = b.x - a.x, dy = b.y - a.y, min = a.r + b.r, d2 = dx * dx + dy * dy;
            if (d2 >= min * min) continue;
            var d = Math.sqrt(d2) || 0.01;
            if (a.lv === b.lv && !a.m && !b.m) { a.m = b.m = true; merges.push([a, b]); continue; }
            var ov = (min - d), ma = a.r * a.r, mb = b.r * b.r, t = ma + mb;
            var nx = dx / d, ny = dy / d;
            a.x -= nx * ov * (mb / t); a.y -= ny * ov * (mb / t);
            b.x += nx * ov * (ma / t); b.y += ny * ov * (ma / t);
          }
        }
        for (i = 0; i < fruits.length; i++) {
          a = fruits[i];
          if (a.x < a.r) { a.x = a.r; }
          if (a.x > W - a.r) { a.x = W - a.r; }
          if (a.y > H - a.r) { a.y = H - a.r; a.px += (a.x - a.px) * 0.25; } // 바닥 마찰
        }
      }
      if (merges.length) {
        merges.forEach(function (p) {
          var A = p[0], B = p[1];
          fruits = fruits.filter(function (f) { return f !== A && f !== B; });
          var lv = A.lv;
          if (lv >= FRUITS.length - 1) { score += 100; return; } // 수박+수박 = 펑!
          var nl = lv + 1, mx = (A.x + B.x) / 2, my = (A.y + B.y) / 2, r = FRUITS[nl].r;
          mx = clampX(mx, r); my = Math.min(H - r, my);
          fruits.push({ id: idSeq++, x: mx, y: my, px: mx, py: my, r: r, lv: nl, born: performance.now() - 600, pop: 8 });
          score += SCORE[nl]; maxLv = Math.max(maxLv, nl);
        });
        onScore && onScore(score);
      }
    }
  }

  function checkOver(dt) {
    var now = performance.now(), danger = false;
    for (var i = 0; i < fruits.length; i++) {
      var f = fruits[i];
      if (now - f.born > 1500 && f.y - f.r < LINE_Y) { danger = true; break; }
    }
    dangerT = danger ? dangerT + dt : 0;
    if (dangerT > 1800) gameOver();
  }

  function gameOver() {
    over = true;
    document.getElementById('g-final').textContent = score;
    document.getElementById('g-over').classList.remove('hidden');
    onOver && onOver(score, maxLv);
  }

  function draw() {
    ctx.fillStyle = '#f4e6c4'; ctx.fillRect(0, 0, W, H);
    ctx.fillStyle = '#eddcb2';
    for (var y = 0; y < H; y += 8) for (var x = (y / 8) % 2 ? 0 : 4; x < W; x += 8) ctx.fillRect(x, y, 4, 4);
    // 데드라인
    var blink = dangerT > 0 && Math.floor(performance.now() / 150) % 2;
    ctx.fillStyle = blink ? '#e43b44' : '#c09a6b';
    for (var lx = 0; lx < W; lx += 4) ctx.fillRect(lx, LINE_Y, 2, 1);
    // 조준선 + 들고있는 과일
    if (!over && cooldown <= 0) {
      var r = FRUITS[cur].r, ax = Math.round(clampX(aimX, r));
      ctx.fillStyle = 'rgba(26,28,44,.25)';
      for (var gy = HOLD_Y + r; gy < H; gy += 4) ctx.fillRect(ax, gy, 1, 2);
      ctx.drawImage(sprites[cur], ax - r, HOLD_Y - r);
    }
    fruits.forEach(function (f) {
      var s = sprites[f.lv];
      if (f.pop > 0) {
        var k = 1 + f.pop * 0.02, w = Math.round(f.r * 2 * k);
        ctx.drawImage(s, Math.round(f.x - w / 2), Math.round(f.y - w / 2), w, w);
        f.pop--;
      } else ctx.drawImage(s, Math.round(f.x - f.r), Math.round(f.y - f.r));
    });
  }

  function loop(t) {
    if (!running) return;
    var dt = Math.min(50, t - (last || t)); last = t;
    if (!over) {
      var n = Math.max(1, Math.round(dt / 16.67));
      for (var k = 0; k < n; k++) { step(); if (cooldown > 0) cooldown--; }
      checkOver(dt);
    }
    draw();
    raf = requestAnimationFrame(loop);
  }

  function start() { if (running) return; running = true; last = 0; raf = requestAnimationFrame(loop); }
  function stop() { running = false; cancelAnimationFrame(raf); }

  return { init: init, start: start, stop: stop, reset: reset, FRUITS: FRUITS };
})();
</script>

        return '<button class="btn btn-s ' + (x.ym === S.walletYm ? 'btn-blue' : 'btn-dark') + '" data-action="wallet-month" data-ym="' + x.ym + '">' + (+x.ym.slice(5)) + '월 도장</button>';
      }).join('') + '</div>' : '') +
      '<div class="w-big"><span>' + (+w.ym.slice(5)) + '월 보유 도장</span><b>' + w.balance + '개</b></div>' +
      '<div class="w-sub">승인 도장 ' + w.approved + '개 · 교환에 사용 ' + w.used + '개 · 출석 ' + w.approved + '/' + w.days + '일</div>' +
      '<div class="board">' + board + '</div>';

    $('cp-rewards').innerHTML = d.rewards.map(function (r) {
      var ok, prog, info, label = '교환';
      if (r.perfect) {
        var claimed = w.claimed.indexOf(r.id) >= 0;
        ok = w.perfect && !claimed; prog = Math.min(1, w.approved / w.days);
        info = '만근 ' + w.approved + '/' + w.days + '일' + (claimed ? ' · 수령 완료' : '');
        if (claimed) label = '받음';
      } else {
        ok = w.balance >= r.cost; prog = Math.min(1, w.balance / r.cost);
        info = '도장 ' + r.cost + '개 사용' + (ok ? '' : ' · ' + (r.cost - w.balance) + '개 부족');
      }
      return '<div class="reward px-box' + (ok ? ' ok' : '') + '">' +
        '<div class="rw-ico">' + Sprites.get(r.icon) + '</div>' +
        '<div class="rw-body"><div class="rw-name">' + esc(r.name) + '</div><div class="rw-desc">' + esc(r.desc) + '</div>' +
        '<div class="rw-desc">' + info + '</div><div class="rw-bar"><i style="width:' + Math.round(prog * 100) + '%"></i></div></div>' +
        '<button class="btn btn-s ' + (ok ? 'btn-green' : 'btn-dark') + '" data-action="exchange" data-id="' + esc(r.id) + '"' + (ok ? '' : ' disabled') + '>' + label + '</button></div>';
    }).join('');
  }

  function cpTab(t) {
    document.querySelectorAll('[data-cp-tab]').forEach(function (b) { b.classList.toggle('active', b.getAttribute('data-cp-tab') === t); });
    ['have', 'shop', 'used'].forEach(function (k) { $('cp-' + k).classList.toggle('hidden', k !== t); });
  }

  function askExchange(id) {
    var r = S.data.rewards.filter(function (x) { return x.id === id; })[0];
    var w = S.data.wallets.filter(function (x) { return x.ym === S.walletYm; })[0];
    if (!r || !w) return;
    openModal('<h3>도장 교환</h3><div class="center"><div class="big-stamp" style="transform:none;width:30%">' + Sprites.get(r.icon) + '</div>' +
      '<p><b>' + esc(r.name) + '</b><br><small>' + esc(r.desc) + '</small></p>' +
      (r.perfect ? '<p class="hint">' + (+w.ym.slice(5)) + '월 만근 보상이에요. 도장은 줄어들지 않아요.</p>'
        : '<p>' + (+w.ym.slice(5)) + '월 도장 <b>' + w.balance + '개 → ' + (w.balance - r.cost) + '개</b></p><p class="hint">교환한 도장은 사라지고, 남은 도장부터 다시 모아요.</p>') +
      '</div><div class="actions"><button class="btn btn-dark" data-action="close">취소</button>' +
      '<button class="btn btn-green" data-action="do-exchange" data-id="' + esc(id) + '">교환하기</button></div>');
  }

  function doExchange(id) {
    var keepYm = S.ym;
    call('exchangeReward', S.name, id, S.walletYm).then(function (data) {
      var c = data.newCoupon;
      // 달력은 보던 달 유지
      if (keepYm !== data.ym) { data.records = S.data.records; data.ym = keepYm; }
      applyMonth(data);
      cpTab('have');
      openModal('<h3>🎉 쿠폰 발급 완료!</h3><div class="coupon-list">' +
        couponHTML({ code: c.code, rewardName: c.rewardName, desc: c.desc, icon: c.icon, ym: c.ym, issuedAt: '방금', used: false }) + '</div>' +
        (c.cost ? '<p class="hint">' + (+c.ym.slice(5)) + '월 남은 도장 ' + c.balanceAfter + '개</p>' : '') +
        '<div class="actions"><button class="btn btn-green" data-action="close">확인</button></div>');
    });
  }

  function couponHTML(c, admin) {
    return '<div class="coupon ' + (c.used ? 'used' : '') + '">' +
      '<div class="c-ico">' + Sprites.get(c.icon || 'ticket') + '</div>' +
      '<div class="c-body"><div class="c-code">' + esc(c.rewardName) + '</div>' +
      '<div class="c-meta">' + (admin === true ? '<b>' + esc(c.name) + '</b> · ' : '') + esc(c.desc) + '</div>' +
      '<div class="c-meta">코드 <b>' + esc(c.code) + '</b></div>' +
      '<div class="c-meta">' + (c.ym ? (+c.ym.slice(5)) + '월 도장 · ' : '') + '발급 ' + esc(c.issuedAt) + (c.used ? ' · 사용 ' + esc(c.usedAt) : '') + '</div></div>' +
      (admin === true ? '<button class="btn btn-s ' + (c.used ? 'btn-dark' : 'btn-red') + '" data-action="use-coupon" data-code="' + esc(c.code) + '">' + (c.used ? '되돌리기' : '사용처리') + '</button>' : '') +
      '<span class="c-tag">' + (c.used ? '사용완료' : '사용가능') + '</span></div>';
  }

  function userPane(id) {
    document.querySelectorAll('#scr-user .pane').forEach(function (p) { p.classList.toggle('active', p.id === id); });
    document.querySelectorAll('.nav-btn').forEach(function (b) { b.classList.toggle('active', b.getAttribute('data-pane') === id); });
    if (id === 'pane-game') {
      if (!S.gameReady) {
        FruitGame.init({
          onScore: function (s) { $('g-score').textContent = s; },
          onOver: function (score, maxLv) {
            $('g-saved').textContent = '기록 저장 중...';
            quiet('saveGameScore', S.name, score, maxLv).then(function (r) {
              $('g-best').textContent = r.myBest;
              var rank = r.top.findIndex(function (x) { return x.name === S.name; });
              $('g-saved').textContent = '저장 완료! 최고 ' + r.myBest + '점' + (rank >= 0 ? ' · 전체 ' + (rank + 1) + '위' : '');
            }).catch(function () { $('g-saved').textContent = '저장 실패 (네트워크 확인)'; });
          },
        });
        S.gameReady = true;
        quiet('getGameRanking', S.name).then(function (r) { $('g-best').textContent = r.myBest; });
      }
      FruitGame.start();
    } else if (S.gameReady) FruitGame.stop();
  }

  function showRanking() {
    call('getGameRanking', S.name).then(function (r) {
      openModal('<h3>🏆 과일 합치기 TOP 10</h3>' +
        (r.top.length ? '<table class="tbl"><tr><th>#</th><th>이름</th><th>최고점</th></tr>' +
          r.top.map(function (x, i) { return '<tr' + (x.name === S.name ? ' style="background:#ffcd75"' : '') + '><td>' + (i + 1) + '</td><td>' + esc(x.name) + '</td><td>' + x.score + '</td></tr>'; }).join('') + '</table>'
          : '<p class="empty" style="color:var(--muted)">아직 기록이 없어요.</p>') +
        '<p class="hint">내 최고점 ' + r.myBest + ' · 플레이 ' + r.myPlays + '회</p>' +
        '<div class="actions"><button class="btn btn-dark" data-action="close">닫기</button></div>');
    });
  }

  /* ── 관리자 화면 ── */
  function adminTab(t) {
    document.querySelectorAll('[data-admin-tab]').forEach(function (b) { b.classList.toggle('active', b.getAttribute('data-admin-tab') === t); });
    document.querySelectorAll('#scr-admin .pane').forEach(function (p) { p.classList.toggle('active', p.id === 'apane-' + t); });
    if (t === 'rec') adminSearch();
    if (t === 'grant') {
      if (!$('g-date').value) $('g-date').value = S.today;
      $('g-date').max = S.today;
      quiet('adminListUsers', S.admin).then(function (list) {
        $('g-users').innerHTML = list.map(function (n) { return '<option value="' + esc(n) + '">'; }).join('');
      }).catch(function () {});
    }
    if (t === 'cp') call('adminListCoupons', S.admin).then(renderAdminCoupons);
    if (t === 'game') call('adminListGames', S.admin).then(renderAdminGames);
  }

  function adminSearch() {
    var opts = { status: $('f-status').value, date: $('f-date').value, name: $('f-name').value };
    call('adminListRecords', S.admin, opts).then(function (res) {
      $('rec-count').textContent = '총 ' + res.total + '건' + (res.total > res.list.length ? ' (최근 ' + res.list.length + '건 표시)' : '') + ' · 기준 ' + res.threshold;
      $('rec-list').innerHTML = res.list.length ? res.list.map(function (r) {
        return '<div class="rec px-box">' +
          '<div class="r-st">' + (r.status === '대기' ? Sprites.get('hourglass') : stampHTML(r.status)) + '</div>' +
          '<div class="r-main"><div class="r-name">' + esc(r.name) + '</div>' +
          '<div class="r-date">' + fmtDate(r.date) + ' · ' + esc(r.status) + (r.admin ? ' (' + esc(r.admin) + ')' : '') + '</div>' +
          (r.granted ? '<div class="r-date" style="color:var(--green-d)">★ ' + esc(r.note) + '</div>' : '') + '</div>' +
          '<span class="r-val ' + (r.pass ? 'pass' : 'fail') + '">' + (r.value == null ? '★' : r.value) + '</span>' +
          '<div class="r-btns">' +
          '<button class="btn btn-green" data-action="review" data-d="승인" data-date="' + r.date + '" data-name="' + esc(r.name) + '"' + (r.status === '승인' ? ' disabled' : '') + '>승인</button>' +
          '<button class="btn btn-red" data-action="review" data-d="비승인" data-date="' + r.date + '" data-name="' + esc(r.name) + '"' + (r.status === '비승인' ? ' disabled' : '') + '>비승인</button>' +
          '</div></div>';
      }).join('') : '<div class="empty">조건에 맞는 기록이 없어요.</div>';
    });
  }

  function review(date, name, decision) {
    call('adminReview', S.admin, date, name, decision).then(function (res) {
      toast(name + ' · ' + decision + ' 처리 완료');
      adminSearch();
    });
  }

  function renderAdminCoupons(res) {
    $('acp-list').innerHTML = res.list.length ? res.list.map(function (c) { return couponHTML(c, true); }).join('') : '<div class="empty">발급된 쿠폰이 없어요.</div>';
  }

  function renderAdminGames(res) {
    $('ag-list').innerHTML =
      '<h3 class="sec-title">🏆 최고점 순위</h3>' +
      (res.ranking.length ? '<table class="tbl"><tr><th>#</th><th>이름</th><th>최고점</th></tr>' + res.ranking.map(function (x, i) { return '<tr><td>' + (i + 1) + '</td><td>' + esc(x.name) + '</td><td>' + x.score + '</td></tr>'; }).join('') + '</table>' : '<div class="empty">기록 없음</div>') +
      '<h3 class="sec-title">🕹 최근 플레이</h3>' +
      (res.list.length ? '<table class="tbl"><tr><th>일시</th><th>이름</th><th>점수</th><th>과일</th></tr>' + res.list.map(function (x) { return '<tr><td>' + esc(x.at.slice(5)) + '</td><td>' + esc(x.name) + '</td><td>' + x.score + '</td><td>' + esc((FruitGame.FRUITS[x.maxLevel] || {}).n || '-') + '</td></tr>'; }).join('') + '</table>' : '<div class="empty">기록 없음</div>');
  }

  /* ── 이벤트 위임 ── */
  document.addEventListener('click', function (e) {
    var el = e.target.closest('[data-action]');
    if (el && !el.disabled) {
      var a = el.getAttribute('data-action');
      if (a === 'logout') logout();
      else if (a === 'prev-month') shiftMonth(-1);
      else if (a === 'next-month') shiftMonth(1);
      else if (a === 'input') openInput(el.getAttribute('data-date'));
      else if (a === 'close') closeModal();
      else if (a === 'save') saveInput(el.getAttribute('data-date'));
      else if (a === 'quick') { var inp = $('m-val'); inp.value = Math.max(0, (Number(inp.value) || 0) + Number(el.getAttribute('data-n'))); judge(); }
      else if (a === 'game-restart') FruitGame.reset();
      else if (a === 'game-rank') showRanking();
      else if (a === 'admin-search') adminSearch();
      else if (a === 'review') review(el.getAttribute('data-date'), el.getAttribute('data-name'), el.getAttribute('data-d'));
      else if (a === 'goto-coupon') { userPane('pane-coupon'); cpTab('have'); }
      else if (a === 'wallet-month') { S.walletYm = el.getAttribute('data-ym'); renderCoupons(); }
      else if (a === 'exchange') askExchange(el.getAttribute('data-id'));
      else if (a === 'do-exchange') doExchange(el.getAttribute('data-id'));
      else if (a === 'use-coupon') call('adminUseCoupon', S.admin, el.getAttribute('data-code')).then(renderAdminCoupons);
      return;
    }
    var nav = e.target.closest('[data-pane]');
    if (nav) return userPane(nav.getAttribute('data-pane'));
    var ct = e.target.closest('[data-cp-tab]');
    if (ct) return cpTab(ct.getAttribute('data-cp-tab'));
    var at = e.target.closest('[data-admin-tab]');
    if (at) return adminTab(at.getAttribute('data-admin-tab'));
    if (e.target.id === 'modal') closeModal();
  });
  $('form-grant').onsubmit = function (e) {
    e.preventDefault();
    var name = $('g-user').value.trim(), date = $('g-date').value, value = $('g-value').value, memo = $('g-memo').value;
    if (!name) return toast('사용자를 입력하세요.');
    if (!date) return toast('날짜를 선택하세요.');
    call('adminGrantStamp', S.admin, name, date, value, memo).then(function (res) {
      var w = res.wallet;
      $('g-result').innerHTML = '<div class="rec px-box"><div class="r-st">' + Sprites.get('stamp-pass') + '</div>' +
        '<div class="r-main"><div class="r-name">' + esc(name) + '</div><div class="r-date">' + fmtDate(date) + ' 합격 도장 ' + (res.existed ? '(기존 기록 승인)' : '(새 기록)') + '</div>' +
        '<div class="r-date">' + (+w.ym.slice(5)) + '월 보유 도장 ' + w.balance + '개 · 출석 ' + w.approved + '/' + w.days + '일</div></div></div>';
      $('g-value').value = ''; $('g-memo').value = '';
      toast(name + ' · ' + fmtDate(date) + ' 합격 도장 완료!');
    });
  };
  $('f-status').onchange = adminSearch;
  $('f-date').onchange = adminSearch;

  /* ── 시작 ── */
  $('logo').innerHTML = Sprites.get('logo');
  Sprites.hydrate();
  var saved = store.get('dot_user');
  if (saved) $('in-user-name').value = saved;
  quiet('getBootstrap').then(function (b) {
    S.threshold = b.threshold; S.today = b.today;
    $('boot-th').textContent = b.threshold;
  }).catch(function () {});
})();
</script>
